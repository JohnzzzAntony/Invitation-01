    26→  // params[0] may be "<slug>" or "<slug>?token=abc" — parse query ourselves.
    27→  function parseSlug(raw) {
    28→    raw = String(raw || '');
    29→    var qi = raw.indexOf('?');
    30→    var slug = qi >= 0 ? raw.slice(0, qi) : raw;
    31→    var token = null;
    32→    if (qi >= 0) {
    33→      raw.slice(qi + 1).split('&').forEach(function (pair) {
    34→        var kv = pair.split('=');
    35→        if (kv[0] === 'token' && kv[1]) {
    36→          try { token = decodeURIComponent(kv[1]); } catch (e) { token = kv[1]; }
    37→        }
    38→      });
    39→    }
    40→    return { slug: slug, token: token };
    41→  }

   295→  function renderSections(ev) {
   296→    var out = '';
   297→    var secs = (ev.sections || []).filter(function (s) { return s && s.visible !== false; });
   298→    secs.forEach(function (s) {
   299→      switch (s.type) {
   300→        case 'hero': out += rHero(s); break;
   301→        case 'text': out += rText(s); break;
   302→        case 'story': out += rStory(s); break;
   303→        case 'countdown': out += rCountdown(s); break;
   304→        case 'details': out += rDetails(s, ev); break;
   305→        case 'schedule': out += rSchedule(s, ev); break;
   306→        case 'gallery': out += rGallery(s, ev); break;
   307→        case 'map': out += rMap(s, ev); break;
   308→        case 'accommodation': out += rAccommodation(s, ev); break;
   309→        case 'registry': out += rRegistry(s, ev); break;
   310→        case 'rsvp': out += rRsvp(s); break;
   311→        case 'contact': out += rContact(s, ev); break;
   312→        case 'divider': out += rDivider(s); break;
   313→        case 'footer': out += rFooter(s); break;
   314→        default: break; // unknown type → skip
   315→      }
   316→    });
   317→    return out;
   318→  }

   585→  function qField(q) {

   761→  function submitRsvp() {
   762→    if (flow.busy) return;
   763→    flow.busy = true;
   764→    var btn = App.qs('[data-fl-submit]');
   765→    if (btn) { btn.disabled = true; btn.innerHTML = '<span class="spinner" style="width:16px;height:16px;border-width:2px"></span> Submitting…'; }
   766→    var body = {
   767→      name: flow.name.trim(),
   768→      email: flow.email.trim(),
   769→      attending: flow.attending === true,
   770→      partySize: flow.attending ? flow.partySize : 0,
   771→      answers: flow.answers,
   772→      subEventIds: flow.attending ? flow.subEventIds : []
   773→    };
   774→    if (flow.token) body.token = flow.token;
   775→    App.api('/api/events/' + encodeURIComponent(flow.ev.id) + '/rsvp', { method: 'POST', body: body })
   776→      .then(function () {
   777→        flow.busy = false;
   778→        var host = App.qs('[data-flow]');
   779→        var foot = App.qs('[data-flow-foot]');
   780→        if (host) host.innerHTML = successHtml();
   781→        if (foot) foot.innerHTML = '';

   783→      .catch(function (err) {

   790→  function icsStamp(d) {
   791→    function p(n) { return String(n).padStart(2, '0'); }
   792→    return '' + d.getFullYear() + p(d.getMonth() + 1) + p(d.getDate()) + 'T' + p(d.getHours()) + p(d.getMinutes()) + p(d.getSeconds());
   793→  }
   794→  function icsText(s) {
   795→    return String(s || '').replace(/\\/g, '\\\\').replace(/([,;])/g, '\\$1').replace(/\n/g, '\\n');
   796→  }

   798→  function downloadIcs() {
   799→    var ev = flow.ev || {};
   800→    var start = new Date(ev.eventDate + 'T' + (ev.startTime || '17:00') + ':00');
   801→    if (isNaN(start.getTime())) start = new Date(ev.eventDate + 'T12:00:00');
   802→    var end = ev.endTime ? new Date(ev.eventDate + 'T' + ev.endTime + ':00') : new Date(start.getTime() + 4 * 3600000);
   803→    if (isNaN(end.getTime()) || end <= start) end = new Date(start.getTime() + 4 * 3600000);
   804→    var ics = [
   805→      'BEGIN:VCALENDAR',
   806→      'VERSION:2.0',
   807→      'PRODID:-//Online RSVP//EN',
   808→      'BEGIN:VEVENT',
   809→      'UID:' + (ev.id || 'event') + '-' + Date.now() + '@onlinersvp',
   810→      'DTSTAMP:' + new Date().toISOString().replace(/[-:]/g, '').replace(/\.\d{3}/, ''),
   811→      'DTSTART:' + icsStamp(start),
   812→      'DTEND:' + icsStamp(end),
   813→      'SUMMARY:' + icsText(ev.name || 'Event'),
   814→      'LOCATION:' + icsText(ev.venue || ''),
   815→      'DESCRIPTION:' + icsText('You are invited: ' + (ev.name || '') + (ev.address ? ' — ' + ev.address : '')),
   816→      'END:VEVENT',
   817→      'END:VCALENDAR'
   818→    ].join('\r\n');
   819→    var blob = new Blob([ics], { type: 'text/calendar;charset=utf-8' });
   820→    var url = URL.createObjectURL(blob);
   821→    var a = document.createElement('a');
   822→    a.href = url;
   823→    a.download = (ev.slug || 'event') + '.ics';
   824→    a.click();
   825→    URL.revokeObjectURL(url);
   826→    App.toast('Calendar file downloaded', 'success');
   827→  }
