   169→  function saveCustomTemplate(t) {
   170→    var arr = readCustom();
   171→    t.id = t.id || ('custom-' + Date.now());
   172→    t.custom = true;
   173→    t.category = 'Custom';
   174→    var found = false;
   175→    for (var i = 0; i < arr.length; i++) {
   176→      if (arr[i].id === t.id) { arr[i] = t; found = true; break; }
   177→    }
   178→    if (!found) arr.push(t);
   179→    writeCustom(arr);
   180→    return t;
   181→  }
   182→  function deleteCustomTemplate(id) {
   183→    writeCustom(readCustom().filter(function (t) { return t.id !== id; }));

   610→  function tickCountdowns(root) {
   611→    var timers = (root || document).querySelectorAll('[data-ws-deadline]');
   612→    Array.prototype.forEach.call(timers, function (t) {
   613→      var end = new Date(t.getAttribute('data-ws-deadline')).getTime();
   614→      if (isNaN(end)) return;
   615→      var diff = Math.max(0, end - Date.now());
   616→      var d = Math.floor(diff / 864e5),
   617→          h = Math.floor(diff % 864e5 / 36e5),
   618→          m = Math.floor(diff % 36e5 / 6e4),
   619→          s = Math.floor(diff % 6e4 / 1e3);
   620→      var map = { d: d, h: h, m: m, s: s };
   621→      for (var u in map) {
   622→        var node = t.querySelector('[data-u="' + u + '"]');
   623→        if (node) node.textContent = String(map[u]).length < 2 ? '0' + map[u] : String(map[u]);
   624→      }
   625→    });
   626→  }

   628→  function bindSite(root) {
   629→    if (!root || root.__wsBound) return;
   630→    root.__wsBound = true;
   631→    /* gentle scroll-reveal for site sections (JS-added class: no-JS visitors
   632→       never see hidden content; honours prefers-reduced-motion) */
   633→    var reduceMotion = false;
   634→    try { reduceMotion = window.matchMedia('(prefers-reduced-motion: reduce)').matches; } catch (e) { /* ignore */ }
   635→    if (!reduceMotion && 'IntersectionObserver' in window) {
   636→      var secs = root.querySelectorAll('.ws-sec, .ws-footer');
   637→      Array.prototype.forEach.call(secs, function (sec) { sec.classList.add('ws-anim'); });
   638→      var secIO = new IntersectionObserver(function (entries) {
   639→        entries.forEach(function (en) {
   640→          if (en.isIntersecting) { en.target.classList.add('ws-in'); secIO.unobserve(en.target); }
   641→        });
   642→      }, { threshold: 0.12 });
   643→      Array.prototype.forEach.call(secs, function (sec) { secIO.observe(sec); });
   644→    }
   645→    /* smooth anchor scrolling inside the preview */
   646→    root.addEventListener('click', function (e) {
   647→      var a = e.target.closest ? e.target.closest('[data-goto]') : null;
   648→      if (!a || !root.contains(a)) return;
   649→      e.preventDefault();
   650→      var target = root.querySelector('#ws-sec-' + a.getAttribute('data-goto'));
   651→      var wrap = root.closest('.ed-canvas-wrap') || document.getElementById('canvas-frame') || root.parentElement;
   652→      if (target && wrap) {
   653→        var top = target.getBoundingClientRect().top - wrap.getBoundingClientRect().top + wrap.scrollTop - 10;
   654→        wrap.scrollTo({ top: Math.max(0, top), behavior: 'smooth' });
   655→      }
   656→    });
   657→    /* RSVP demo submit */
   658→    root.addEventListener('submit', function (e) {
   659→      var form = e.target.closest ? e.target.closest('.ws-rsvp-form') : null;
   660→      if (!form || !root.contains(form)) return;
   661→      e.preventDefault();
   662→      var name = form.querySelector('[name="name"]');
   663→      var attend = form.querySelector('[name="attend"]');
   664→      if (name && !name.value.trim()) { name.classList.add('ws-err'); name.focus(); return; }
   665→      if (attend && !attend.value) { attend.classList.add('ws-err'); attend.focus(); return; }
   666→      var sec = form.closest('.ws-rsvp');
   667→      var doneMsg = root.__wsData && root.__wsData.sections.rsvp.success;
   668→      var box = document.createElement('div');
