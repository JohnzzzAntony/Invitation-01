   360→  App.downloadCsv = function (filename, rows) {
   361→    function escape(v) {
   362→      var s = v == null ? '' : String(v);
   363→      return /[",\n]/.test(s) ? '"' + s.replace(/"/g, '""') + '"' : s;
   364→    }
   365→    var csv = rows.map(function (r) { return r.map(escape).join(','); }).join('\n');
   366→    var blob = new Blob([csv], { type: 'text/csv;charset=utf-8' });
   367→    var url = URL.createObjectURL(blob);
   368→    var a = document.createElement('a');
   369→    a.href = url; a.download = filename; a.click();
   370→    URL.revokeObjectURL(url);
   371→  };
   372→  App.parseCsv = function (text) {
   373→    var rows = [], row = [], field = '', inQ = false;
   374→    for (var i = 0; i < text.length; i++) {
   375→      var c = text[i];
   376→      if (inQ) {
   377→        if (c === '"') { if (text[i + 1] === '"') { field += '"'; i++; } else inQ = false; }
   378→        else field += c;
   379→      } else if (c === '"') inQ = true;
   380→      else if (c === ',') { row.push(field); field = ''; }
   381→      else if (c === '\n' || c === '\r') {
   382→        if (field !== '' || row.length) { row.push(field); rows.push(row); }
   383→        row = []; field = '';
   384→        if (c === '\r' && text[i + 1] === '\n') i++;
   385→      } else field += c;
   386→    }
   387→    if (field !== '' || row.length) { row.push(field); rows.push(row); }
   388→    return rows;
   389→  };

   566→  /* ---------------- Countdown helper ---------------- */
   567→  App.bindCountdown = function (el, targetIso, compact) {
   568→    if (!el) return;
   569→    var target = new Date(targetIso).getTime();
   570→    function tick() {
   571→      var diff = Math.max(0, target - Date.now());
   572→      var d = Math.floor(diff / 86400000);
   573→      var h = Math.floor((diff % 86400000) / 3600000);
   574→      var m = Math.floor((diff % 3600000) / 60000);
   575→      var s = Math.floor((diff % 60000) / 1000);
   576→      var cells = [[d, 'Days'], [h, 'Hours'], [m, 'Minutes'], [s, 'Seconds']];
   577→      el.innerHTML = cells.map(function (c) {
   578→        return '<div class="mc-box"><div class="mc-num">' + String(c[0]).padStart(2, '0') + '</div><div class="mc-lab">' + c[1] + '</div></div>';
   579→      }).join('');
   580→    }
   581→    tick();
   582→    var iv = setInterval(tick, 1000);
   583→    App.addCleanup(function () { clearInterval(iv); });
   584→  };
