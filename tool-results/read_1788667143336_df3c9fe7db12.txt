   122→  function deepMerge(dst, src) {
   123→    if (!src || typeof src !== 'object') return dst;
   124→    for (var k in src) {
   125→      if (!Object.prototype.hasOwnProperty.call(src, k)) continue;
   126→      if (src[k] && typeof src[k] === 'object' && !Array.isArray(src[k]) && dst[k] && typeof dst[k] === 'object' && !Array.isArray(dst[k])) {
   127→        deepMerge(dst[k], src[k]);
   128→      } else if (src[k] !== undefined) {
   129→        dst[k] = src[k];
   130→      }
   131→    }
   132→    return dst;
   133→  }

   227→  function fieldHtml(f, value, onInput) {
   228→    var wrap = document.createElement('div');
   229→    wrap.className = 'ed-field' + (f.type === 'check' ? ' ed-check' : '');
   230→    var id = 'f-' + Math.random().toString(36).slice(2, 8);
   231→    if (f.type === 'check') {
   232→      wrap.innerHTML = '<input type="checkbox" id="' + id + '"' + (value ? ' checked' : '') + '/>' +
   233→        '<label for="' + id + '">' + esc(f.label) + '</label>';
   234→      wrap.querySelector('input').addEventListener('change', function (e) { onInput(e.target.checked); });
   235→      return wrap;
   236→    }
   237→    var label = '<label for="' + id + '">' + esc(f.label) + '</label>';
   238→    var ctl;
   239→    if (f.type === 'date') {
   240→      ctl = '<input id="' + id + '" type="date" value="' + esc(value) + '"/>';
   241→    } else if (f.type === 'number') {
   242→      ctl = '<input id="' + id + '" type="number" min="' + (f.min || 1) + '" max="' + (f.max || 12) + '" value="' + esc(value) + '"/>';
   243→    } else if (f.type === 'photo') {
   244→      var opts = '<option value="">Choose photo…</option>';
   245→      window.EVER_PHOTOS.forEach(function (p) {
   246→        opts += '<option value="' + p.id + '"' + (value === p.id ? ' selected' : '') + '>' + esc(p.label) + '</option>';
   247→      });
   248→      var isUrl = value && String(value).indexOf('/') > -1;
   249→      ctl = '<div class="ed-photo-row">' +
   250→          '<select data-role="pick">' + opts + '</select>' +
   251→          '<input type="text" data-role="url" placeholder="…or paste image URL" value="' + (isUrl ? esc(value) : '') + '"/>' +
   252→        '</div>';
   253→    } else if (f.type === 'icon') {
   254→      var labels = { rings: 'Rings', cocktail: 'Glasses', dinner: 'Dinner', music: 'Music', calendar: 'Calendar',
   255→        clock: 'Clock', pin: 'Pin', dress: 'Dress', car: 'Parking', bell: 'Bell', chair: 'Accessible',
   256→        snow: 'Cooling', phone: 'Phone', mail: 'Mail', whatsapp: 'WhatsApp', facebook: 'Facebook',
   257→        instagram: 'Instagram', share: 'Share', heart: 'Heart', camera: 'Camera' };
   258→      var iopts = '';
   259→      Object.keys(window.EVER_ICONS).forEach(function (k) {
   260→        iopts += '<option value="' + k + '"' + (value === k ? ' selected' : '') + '>' + (labels[k] || k) + '</option>';
   261→      });
   262→      ctl = '<select data-role="icon">' + iopts + '</select>';
   263→    } else {
   264→      ctl = '<input id="' + id + '" type="text" placeholder="' + esc(f.ph || '') + '" value="' + esc(value) + '"/>';
   265→    }
   266→    wrap.innerHTML = label + ctl;
   267→
   268→    if (f.type === 'photo') {
   269→      var pick = wrap.querySelector('[data-role="pick"]');
   270→      var url = wrap.querySelector('[data-role="url"]');
   271→      pick.addEventListener('change', function () {
   272→        if (!pick.value) return;
   273→        url.value = '';
   274→        onInput(pick.value);
   275→      });
   276→      url.addEventListener('input', function () {
   277→        onInput(url.value.trim());
   278→      });
   279→    } else if (f.type === 'icon') {
   280→      wrap.querySelector('select').addEventListener('change', function (e) { onInput(e.target.value); });
   281→    } else if (f.type !== 'check') {
   282→      var txt = wrap.querySelector('input');
   283→      txt.addEventListener('input', function (e) { onInput(e.target.value); });
   284→      /* UX guard for link fields: flag anything that is not an http(s) URL.
   285→         (The renderer also enforces this — see safeUrl() in templates.js.) */
   286→      if (/Url$/.test(f.k || '')) {
   287→        txt.addEventListener('blur', function () {
   288→          var v = txt.value.trim();
   289→          var bad = v !== '' && !/^(https?:\/\/|#[a-z0-9_-]*)/i.test(v);
   290→          txt.classList.toggle('ed-invalid', bad);
   291→          txt.title = bad ? 'Use a full link starting with https://' : '';
   292→        });
   293→      }
   294→    }
   295→    return wrap;
   296→  }

   298→  function getVal(obj, path) {
   299→    return path.split('.').reduce(function (o, k) { return o ? o[k] : undefined; }, obj);
   300→  }
   301→  function setVal(obj, path, val) {
   302→    var ks = path.split('.');
   303→    var o = obj;
   304→    for (var i = 0; i < ks.length - 1; i++) {
   305→      if (!o[ks[i]] || typeof o[ks[i]] !== 'object') o[ks[i]] = {};
   306→      o = o[ks[i]];
   307→    }
   308→    o[ks[ks.length - 1]] = val;
   309→  }

   314→  function buildGroups() {
   315→    var host = document.getElementById('sec-groups');
   316→    if (!host) return;
   317→    host.innerHTML = '';
   318→    var order = state.order && state.order.length ? state.order : SECTIONS.map(function (s) { return s.id; });
   319→
   320→    order.forEach(function (sid) {
   321→      var meta = null;
   322→      SECTIONS.forEach(function (s) { if (s.id === sid) meta = s; });
   323→      if (!meta) return;
   324→      var sec = state.sections[sid];
   325→
   326→      var det = document.createElement('details');
   327→      det.className = 'ed-group' + (ui.open[sid] ? ' open' : '') + (sec.on ? '' : ' sec-off');
   328→      if (ui.open[sid]) det.open = true;
   329→
   330→      var sum = document.createElement('summary');
   331→      sum.innerHTML = '<span>' + esc(meta.label) + '</span>' +
   332→        '<span class="sec-tools">' +
   333→          '<button type="button" data-tool="up" aria-label="Move ' + esc(meta.label) + ' up">&#8593;</button>' +
   334→          '<button type="button" data-tool="down" aria-label="Move ' + esc(meta.label) + ' down">&#8595;</button>' +
   335→          '<button type="button" data-tool="eye" aria-label="Show or hide ' + esc(meta.label) + '" aria-pressed="' + sec.on + '">&#128065;</button>' +
   336→        '</span>';
   337→
   338→      var body = document.createElement('div');
   339→      body.className = 'ed-body';
   340→
   341→      meta.fields.forEach(function (f) {
   342→        body.appendChild(fieldHtml(f, getVal(sec, f.k), function (v) {
   343→          setVal(sec, f.k, v);
   344→          refresh();
   345→        }));
   346→      });
   347→
   348→      if (meta.list) buildList(body, meta, sec);
   349→
   350→      /* tool buttons */
   351→      sum.querySelectorAll('[data-tool]').forEach(function (btn) {
   352→        btn.addEventListener('click', function (e) {
   353→          e.preventDefault();
   354→          e.stopPropagation();
   355→          var tool = btn.getAttribute('data-tool');
   356→          var i = state.order.indexOf(sid);
   357→          if (tool === 'eye') {
   358→            sec.on = !sec.on;
   359→            det.classList.toggle('sec-off', !sec.on);
   360→            btn.setAttribute('aria-pressed', String(sec.on));
   361→          } else if (tool === 'up' && i > 0) {
   362→            state.order.splice(i, 1);
   363→            state.order.splice(i - 1, 0, sid);
   364→            buildGroups();
   365→          } else if (tool === 'down' && i < state.order.length - 1) {
   366→            state.order.splice(i, 1);
   367→            state.order.splice(i + 1, 0, sid);
   368→            buildGroups();
   369→          }
   370→          refresh();
   371→        });
   372→      });
   373→      det.addEventListener('toggle', function () { ui.open[sid] = det.open; });
   374→
   375→      det.appendChild(sum);
   376→      det.appendChild(body);
   377→      host.appendChild(det);
   378→    });
   379→  }

   381→  function buildList(body, meta, sec) {
   382→    var cfg = meta.list;
   383→    var arr = sec[cfg.k] = Array.isArray(sec[cfg.k]) ? sec[cfg.k] : [];
   384→    var listBox = document.createElement('div');
   385→    listBox.className = 'ed-list';
   386→
   387→    arr.forEach(function (item, idx) {
   388→      var li = document.createElement('div');
   389→      li.className = 'ed-li';
   390→      var head = document.createElement('div');
   391→      head.className = 'ed-li-head';
   392→      head.innerHTML = '<b>' + esc(cfg.label) + ' ' + (idx + 1) + '</b>' +
   393→        '<span>' +
   394→          '<button type="button" data-op="up" aria-label="Move up" ' + (idx === 0 ? 'disabled' : '') + '>&#8593;</button>' +
   395→          '<button type="button" data-op="down" aria-label="Move down" ' + (idx === arr.length - 1 ? 'disabled' : '') + '>&#8595;</button>' +
   396→          '<button type="button" data-op="del" aria-label="Remove">&#10005;</button>' +
   397→        '</span>';
   398→      li.appendChild(head);
   399→
   400→      cfg.fields.forEach(function (f) {
   401→        var val = f.k ? (item[f.k] || '') : item;
   402→        li.appendChild(fieldHtml(f, val, function (v) {
   403→          if (f.k) item[f.k] = v; else sec[cfg.k][idx] = v;
   404→          refresh();
   405→        }));
   406→      });
   407→
   408→      head.querySelectorAll('[data-op]').forEach(function (btn) {
   409→        btn.addEventListener('click', function () {
   410→          var op = btn.getAttribute('data-op');
   411→          if (op === 'del') arr.splice(idx, 1);
   412→          if (op === 'up' && idx > 0) arr.splice(idx - 1, 0, arr.splice(idx, 1)[0]);
   413→          if (op === 'down' && idx < arr.length - 1) arr.splice(idx + 1, 0, arr.splice(idx, 1)[0]);
   414→          buildGroups();
   415→          refresh();
   416→        });
   417→      });
   418→      listBox.appendChild(li);
   419→    });
   420→
   421→    var add = document.createElement('button');
   422→    add.type = 'button';
   423→    add.className = 'ed-add';
   424→    add.textContent = '+ Add ' + cfg.label.toLowerCase();
   425→    add.addEventListener('click', function () {
   426→      arr.push(cfg.blank());
   427→      buildGroups();
   428→      refresh();
   429→    });
   430→    listBox.appendChild(add);
   431→    body.appendChild(listBox);
   432→  }

   598→  function bindChrome() {
   599→    document.querySelectorAll('.ed-tab').forEach(function (tab) {
   600→      tab.addEventListener('click', function () {
   601→        document.querySelectorAll('.ed-tab').forEach(function (t) {
   602→          t.classList.remove('on');
   603→          t.setAttribute('aria-selected', 'false');
   604→        });
   605→        tab.classList.add('on');
   606→        tab.setAttribute('aria-selected', 'true');
   607→        document.querySelectorAll('.ed-panel').forEach(function (p) { p.hidden = true; });
   608→        var panel = document.getElementById(tab.getAttribute('aria-controls'));
   609→        if (panel) panel.hidden = false;
   610→      });
   611→    });
   612→
   613→    document.querySelectorAll('.ed-dev-btn').forEach(function (btn) {
   614→      btn.addEventListener('click', function () {
   615→        document.querySelectorAll('.ed-dev-btn').forEach(function (b) {
   616→          b.classList.remove('on');
   617→          b.setAttribute('aria-pressed', 'false');
   618→        });
   619→        btn.classList.add('on');
   620→        btn.setAttribute('aria-pressed', 'true');
   621→        document.getElementById('canvas-frame').classList.toggle('mobile', btn.getAttribute('data-dev') === 'mobile');
   622→      });
   623→    });
   624→
   625→    document.getElementById('save-btn').addEventListener('click', function () {
   626→      saveNow();
   627→      toast('Saved — your website lives on this device.');
   628→    });
   629→
   630→    document.getElementById('publish-btn').addEventListener('click', function () {
   631→      saveNow();
   632→      var flow = readJson(FLOW_KEY) || {};
   633→      var slug = slugify(state.basics.nameA + '-' + state.basics.nameB);
   634→      flow.published = true;
   635→      flow.slug = slug;
   636→      writeJson(FLOW_KEY, flow);
   637→      var badge = document.getElementById('ed-badge');
   638→      badge.textContent = 'Published';
   639→      badge.classList.add('published');
   640→      var urlEl = document.getElementById('pub-url');
   641→      if (urlEl) urlEl.textContent = slug + '.ever-rsvp.com';
   642→      document.getElementById('pub-overlay').hidden = false;
   643→      document.body.classList.add('modal-open');
   644→    });
   645→    document.getElementById('pub-close').addEventListener('click', function () {
   646→      document.getElementById('pub-overlay').hidden = true;
   647→      document.body.classList.remove('modal-open');
   648→    });
   649→    var copy = document.getElementById('copy-url');
   650→    if (copy) {
   651→      copy.addEventListener('click', function () {
   652→        var text = document.getElementById('pub-url').textContent;
   653→        var done = function () { copy.textContent = 'Copied!'; setTimeout(function () { copy.textContent = 'Copy'; }, 1600); };
   654→        if (navigator.clipboard && navigator.clipboard.writeText) navigator.clipboard.writeText(text).then(done, done);
   655→        else done();
   656→      });
   657→    }
   658→  }
