   348→   348→  function saveCustomTemplate(t) {
   349→   349→    var arr = readCustom();
   350→   350→    t.id = t.id || ('custom-' + Date.now());
   351→   351→    t.custom = true;
   352→   352→    t.event = t.event || 'wedding';
   353→   353→    t.layout = t.layout || 'royal';
   354→   354→    t.category = t.category || 'Custom';
   355→   355→    var found = false;
   356→   356→    for (var i = 0; i < arr.length; i++) {
   357→   357→      if (arr[i].id === t.id) { arr[i] = t; found = true; break; }
   358→   358→    }
   359→   359→    if (!found) arr.push(t);
   360→   360→    writeCustom(arr);
   361→   361→    return t;
   362→   362→  }

   364→   364→    writeCustom(readCustom().filter(function (t) { return t.id !== id; }));

   382→   382→    var tpl = templateId ? findTemplate(templateId) : THEMES[0];
   383→   383→    var layout = findLayout(tpl.layout || 'royal');
   384→   384→    var dflt = layout.defaults ? layout.defaults() : { basics: {}, sections: {} };
   385→   385→    var d = {
   386→   386→      v: 4,
   387→   387→      layoutId: layout.id,
   388→   388→      templateId: tpl.id,
   389→   389→      nameFont: '',            /* '' = follow the template */
   390→   390→      bodyFont: 'lato',
   391→   391→      accent: 'tpl',           /* 'tpl' or hex */
   392→   392→      btnShape: 'soft',
   393→   393→      spacing: 'normal',
   394→   394→      basics: dflt.basics || {},
   395→   395→      order: layoutOrder(layout),
   396→   396→      sections: dflt.sections || {}
   397→   397→    };
   398→   398→    if (tpl.content) {
   399→   399→      if (tpl.content.basics) deepMerge(d.basics, tpl.content.basics);
   400→   400→      if (tpl.content.sections) {
   401→   401→        for (var k in tpl.content.sections) {
   402→   402→          if (!d.sections[k]) d.sections[k] = { on: true };
   403→   403→          deepMerge(d.sections[k], tpl.content.sections[k]);
   404→   404→        }
   405→   405→      }
   406→   406→    }
   407→   407→    return d;
   408→   408→  }

   411→   411→   *  State loader (editor) with v1/v2/v3 → v4 migration                 *
   412→   412→   * ------------------------------------------------------------------ */
   413→   413→  function loadSiteState(flow) {
   414→   414→    flow = flow || {};
   415→   415→    var saved = readJson(EVENT_KEY);
   416→   416→    var wanted = flow.design || (saved && saved.templateId) || '';
   417→   417→    if (wanted && !templateExists(wanted)) wanted = '';
   418→   418→    var d = siteDefaults(wanted || undefined);
   419→   419→
   420→   420→    if (!saved || typeof saved !== 'object') return d;
   421→   421→
   422→   422→    /* theme-ish top-level fields carry over on every version */
   423→   423→    ['nameFont', 'bodyFont', 'accent', 'btnShape', 'spacing'].forEach(function (k) {
   424→   424→      if (saved[k] !== undefined && saved[k] !== null && saved[k] !== '') d[k] = saved[k];
   425→   425→    });
   426→   426→
   427→   427→    var savedLayout = null;
   428→   428→    if (saved.v === 4 && saved.layoutId && LAYOUTS.some(function (l) { return l.id === saved.layoutId; })) {
   429→   429→      savedLayout = saved.layoutId;
   430→   430→    } else if (saved.v === 3) {
   431→   431→      savedLayout = 'royal'; /* v3 sites used the classic wedding format */
   432→   432→    }
   433→   433→
   434→   434→    if (saved.v === 4) {
   435→   435→      /* keep user edits; extra section keys are simply ignored by renderers */
   436→   436→      deepMerge(d, saved);
   437→   437→      if (saved.order && Array.isArray(saved.order) && saved.order.length) d.order = saved.order.slice();
   438→   438→    } else if (saved.v === 3) {
   439→   439→      if (saved.basics) deepMerge(d.basics, saved.basics);
   440→   440→      if (saved.sections) {
   441→   441→        for (var k in saved.sections) {
   442→   442→          if (d.sections[k]) deepMerge(d.sections[k], saved.sections[k]);
   443→   443→        }
   444→   444→      }
   445→   445→      if (Array.isArray(saved.order) && saved.order.length === d.order.length) d.order = saved.order.slice();
   446→   446→    } else {
   447→   447→      /* v1/v2 legacy: carry only universal bits */
   448→   448→      var TMAP = { eucalyptus: 'sage', monogram: 'champagne', 'golden-hour': 'champagne',
   449→   449→        midnight: 'navy', 'garden-party': 'sage', ocean: 'navy', ivory: 'champagne' };
   450→   450→      if (saved.sections && saved.sections.rsvp && saved.sections.rsvp.deadline) {
   451→   451→        d.sections.rsvp.deadline = saved.sections.rsvp.deadline;
   452→   452→      }
   453→   453→      if (saved.nameFont === 'classic') d.nameFont = 'corm';
   454→   454→      if (saved.nameFont === 'romantic') d.nameFont = 'paris';
   455→   455→      if (saved.nameFont === 'modern') d.nameFont = 'jost';
   456→   456→      void TMAP;
   457→   457→    }
   458→   458→
   459→   459→    /* the design picked in checkout wins, but keep edits when the layout matches */
   460→   460→    if (flow.design && templateExists(flow.design) && flow.design !== d.templateId) {
   461→   461→      var newTpl = findTemplate(flow.design);
   462→   462→      if (newTpl.layout === d.layoutId) {
   463→   463→        d.templateId = flow.design; /* keep user edits */
   464→   464→      } else {
   465→   465→        d = siteDefaults(flow.design); /* new event type → fresh sample content */
   466→   466→      }
   467→   467→    }
   468→   468→    if (!templateExists(d.templateId)) d.templateId = THEMES[0].id;
   469→   469→    var tpl = findTemplate(d.templateId);
   470→   470→    if (!LAYOUTS.some(function (l) { return l.id === d.layoutId; }) || d.layoutId !== (tpl.layout || 'royal')) {
   471→   471→      /* switching layouts keeps overlapping content keys via deepMerge below */
   472→   472→      var keep = { basics: d.basics, sections: d.sections };
   473→   473→      d = siteDefaults(d.templateId);
   474→   474→      deepMerge(d.basics, keep.basics);
   475→   475→      for (var sk in keep.sections) {
   476→   476→        if (d.sections[sk]) deepMerge(d.sections[sk], keep.sections[sk]);
   477→   477→      }
   478→   478→    }
   479→   479→    void savedLayout;
   480→   480→    return d;
   481→   481→  }

   581→   581→  function tickCountdowns(root) {
   582→   582→    var timers = (root || document).querySelectorAll('[data-ws-deadline]');
   583→   583→    Array.prototype.forEach.call(timers, function (t) {
   584→   584→      var end = new Date(t.getAttribute('data-ws-deadline')).getTime();
   585→   585→      if (isNaN(end)) return;
   586→   586→      var diff = Math.max(0, end - Date.now());
   587→   587→      var d = Math.floor(diff / 864e5),
   588→   588→          h = Math.floor(diff % 864e5 / 36e5),
   589→   589→          m = Math.floor(diff % 36e5 / 6e4),
   590→   590→          s = Math.floor(diff % 6e4 / 1e3);
   591→   591→      var map = { d: d, h: h, m: m, s: s };
   592→   592→      for (var u in map) {
   593→   593→        var node = t.querySelector('[data-u="' + u + '"]');
   594→   594→        if (node) node.textContent = String(map[u]).length < 2 ? '0' + map[u] : String(map[u]);
   595→   595→      }
   596→   596→    });
   597→   597→  }

   599→   599→  function bindSite(root) {
   600→   600→    if (!root || root.__wsBound) return;
   601→   601→    root.__wsBound = true;
   602→   602→    /* gentle scroll-reveal for site sections (JS-added class: no-JS visitors
   603→   603→       never see hidden content; honours prefers-reduced-motion) */
   604→   604→    var reduceMotion = false;
   605→   605→    try { reduceMotion = window.matchMedia('(prefers-reduced-motion: reduce)').matches; } catch (e) { /* ignore */ }
   606→   606→    if (!reduceMotion && 'IntersectionObserver' in window) {
   607→   607→      var secs = root.querySelectorAll('.ws-sec, .ws-footer');
   608→   608→      Array.prototype.forEach.call(secs, function (sec) { sec.classList.add('ws-anim'); });
   609→   609→      var secIO = new IntersectionObserver(function (entries) {
   610→   610→        entries.forEach(function (en) {
   611→   611→          if (en.isIntersecting) { en.target.classList.add('ws-in'); secIO.unobserve(en.target); }
   612→   612→        });
   613→   613→      }, { threshold: 0.12 });
   614→   614→      Array.prototype.forEach.call(secs, function (sec) { secIO.observe(sec); });
   615→   615→    }
   616→   616→    /* smooth anchor scrolling inside the preview */
   617→   617→    root.addEventListener('click', function (e) {
   618→   618→      var a = e.target.closest ? e.target.closest('[data-goto]') : null;
   619→   619→      if (!a || !root.contains(a)) return;
   620→   620→      e.preventDefault();
   621→   621→      var target = root.querySelector('#ws-sec-' + a.getAttribute('data-goto'));
   622→   622→      var wrap = root.closest('.ed-canvas-wrap') || document.getElementById('canvas-frame') || root.parentElement;
   623→   623→      if (target && wrap) {
   624→   624→        var top = target.getBoundingClientRect().top - wrap.getBoundingClientRect().top + wrap.scrollTop - 10;
   625→   625→        wrap.scrollTo({ top: Math.max(0, top), behavior: 'smooth' });
   626→   626→      }
   627→   627→    });
   628→   628→    /* RSVP demo submit */
   629→   629→    root.addEventListener('submit', function (e) {
   630→   630→      var form = e.target.closest ? e.target.closest('.ws-rsvp-form') : null;
   631→   631→      if (!form || !root.contains(form)) return;
   632→   632→      e.preventDefault();
   633→   633→      var name = form.querySelector('[name="name"]');
   634→   634→      var attend = form.querySelector('[name="attend"]');
   635→   635→      if (name && !name.value.trim()) { name.classList.add('ws-err'); name.focus(); return; }
   636→   636→      if (attend && !attend.value) { attend.classList.add('ws-err'); attend.focus(); return; }
   637→   637→      var doneMsg = root.__wsData && root.__wsData.sections && root.__wsData.sections.rsvp &&
   638→   638→                    root.__wsData.sections.rsvp.success;
   639→   639→      var box = document.createElement('div');
   640→   640→      box.className = 'ws-rsvp-done';
   641→   641→      box.innerHTML = '<span class="ws-done-ic" aria-hidden="true"><svg viewBox="0 0 24 24"><circle cx="12" cy="12" r="10"/><path d="M7.5 12.5l3 3 6-6.5"/></svg></span>' +
   642→   642→        '<b>RSVP received</b><p>' + esc(doneMsg || 'Thank you!') + '</p>';
   643→   643→      form.hidden = true;
   644→   644→      (form.parentElement || form.closest('.ws-sec') || root).appendChild(box);
   645→   645→    });
   646→   646→    root.addEventListener('input', function (e) {
   647→   647→      if (e.target.classList) e.target.classList.remove('ws-err');
   648→   648→    }, true);
   649→   649→  }
