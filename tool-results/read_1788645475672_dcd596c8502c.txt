   168→  function esc(s) {
   169→    return String(s == null ? '' : s)
   170→      .replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;').replace(/"/g, '&quot;');
   171→  }
   172→  function slugify(s) {
   173→    return String(s || '').toLowerCase().replace(/[^a-z0-9]+/g, '-').replace(/^-+|-+$/g, '') || 'your-event';
   174→  }
   175→  function hexToRgba(hex, a) {
   176→    var h = String(hex || '').replace('#', '');
   177→    if (h.length === 3) h = h[0] + h[0] + h[1] + h[1] + h[2] + h[2];
   178→    var n = parseInt(h, 10);
   179→    if (isNaN(n)) return 'rgba(0,0,0,' + a + ')';
   180→    return 'rgba(' + ((n >> 16) & 255) + ',' + ((n >> 8) & 255) + ',' + (n & 255) + ',' + a + ')';
   181→  }

   218→  /* ---------- Load / save (with v1 migration) ---------- */
   219→
   220→  function loadState() {
   221→    var flow = readJson(FLOW_KEY) || {};
   222→    var saved = readJson(EVENT_KEY);
   223→    var d = defaults();
   224→    if (flow.design) d.templateId = flow.design;
   225→
   226→    if (saved && saved.v === 2) {
   227→      for (var k in d) {
   228→        if (!Object.prototype.hasOwnProperty.call(d, k)) continue;
   229→        if (k === 'sections') continue;
   230→        if (Object.prototype.hasOwnProperty.call(saved, k)) d[k] = saved[k];
   231→      }
   232→      if (saved.order && saved.order.length === ORDER.length) d.order = saved.order.slice();
   233→      for (var sid in d.sections) {
   234→        if (!Object.prototype.hasOwnProperty.call(d.sections, sid)) continue;
   235→        if (saved.sections && saved.sections[sid]) {
   236→          for (var fk in d.sections[sid]) {
   237→            if (Object.prototype.hasOwnProperty.call(saved.sections[sid], fk)) {
   238→              d.sections[sid][fk] = saved.sections[sid][fk];
   239→            }
   240→          }
   241→        }
   242→      }
   243→    } else if (saved) {
   244→      /* legacy v1 editor state — carry over what still makes sense */
   245→      ['templateId','fontPair','accent','nameA','nameB','date','venue','city'].forEach(function (k) {
   246→        if (Object.prototype.hasOwnProperty.call(saved, k) && saved[k]) d[k] = saved[k];
   247→      });
   248→      if (saved.story) d.sections.story.text = saved.story;
   249→      if (typeof saved.showStory === 'boolean') d.sections.story.on = saved.showStory;
   250→      if (typeof saved.showSchedule === 'boolean') d.sections.schedule.on = saved.showSchedule;
   251→      ['qMenu','qPlus','qSong'].forEach(function (k) {
   252→        if (typeof saved[k] === 'boolean') d.sections.rsvp[k] = saved[k];
   253→      });
   254→    }
   255→    state = d;
   256→  }

   360→  function onFieldInput(e) {
   361→    var el = e.target;
   362→    var secId = el.getAttribute('data-sec');
   363→    if (el.getAttribute('data-style')) {           /* design panel select */
   364→      state[el.getAttribute('data-style')] = el.value;
   365→      autosave(); render();
   366→      return;
   367→    }
   368→    if (!secId) return;
   369→    var k = el.getAttribute('data-k');
   370→    var val = el.type === 'checkbox' ? el.checked : el.value;
   371→    var i = el.getAttribute('data-i');
   372→
   373→    if (secId === '_basic') {
   374→      state[k] = val;
   375→    } else if (i !== null && i !== undefined) {
   376→      state.sections[secId][k][+i][el.getAttribute('data-fk')] = val;
   377→      var li = el.closest('.ed-li');
   378→      if (li) {
   379→        var b = li.querySelector('.ed-li-head b');
   380→        var item = state.sections[secId][k][+i];
   381→        if (b) b.textContent = (+i + 1) + '. ' + (item.title || item.q || 'Item ' + (+i + 1));
   382→      }
   383→    } else {
   384→      state.sections[secId][k] = val;
   385→    }
   386→    autosave(); render();
   387→  }

   389→  function onSidebarClick(e) {
   390→    var btn = e.target.closest('[data-act]');
   391→    if (!btn) return;
   392→    e.preventDefault();          /* keep summary from toggling when using the tools */
   393→    e.stopPropagation();
   394→    var act = btn.getAttribute('data-act');
   395→    var secId = btn.getAttribute('data-sec');
   396→
   397→    if (act === 'eye') {
   398→      var s = state.sections[secId];
   399→      s.on = !s.on;
   400→      btn.classList.toggle('on', s.on);
   401→      btn.setAttribute('aria-pressed', String(s.on));
   402→      btn.setAttribute('aria-label', (s.on ? 'Hide ' : 'Show ') + secMeta(secId).label + ' section');
   403→      var grp = secGroups.querySelector('details[data-sec="' + secId + '"]');
   404→      if (grp) grp.classList.toggle('off', !s.on);
   405→      autosave(); render();
   406→      return;
   407→    }
   408→
   409→    if (act === 'up' || act === 'down') {
   410→      var idx = state.order.indexOf(secId);
   411→      var to = act === 'up' ? idx - 1 : idx + 1;
   412→      if (to < 0 || to >= state.order.length) return;
   413→      var tmp = state.order[idx];
   414→      state.order[idx] = state.order[to];
   415→      state.order[to] = tmp;
   416→      buildContentPanel();
   417→      autosave(); render();
   418→      return;

   704→  function render() {
   705→    var canvas = document.getElementById('canvas');
   706→    if (!canvas) return;
   707→
   708→    var tpl = findTpl(state.templateId);
   709→    var accent = accentHex();
   710→    var wrap = document.querySelector('.ed-canvas-wrap');
   711→    var scrollPos = wrap ? wrap.scrollTop : 0;
   712→    if (pvTimer) { clearInterval(pvTimer); pvTimer = null; }
   713→
   714→    var root = document.createElement('div');
   715→    root.className = 'pv fm-' + state.fontPair + ' bs-' + state.btnShape + ' sp-' + state.spacing + (tpl.dots ? ' pv-dots' : '');
   716→    root.style.background = tpl.bg;
   717→    root.style.color = tpl.text;
   718→    root.style.setProperty('--pv-accent', accent);
   719→    root.style.setProperty('--pv-btn-text', tpl.btnText);
   720→    if (tpl.dots) {
   721→      root.style.backgroundImage = 'radial-gradient(' + hexToRgba(tpl.text, 0.09) + ' 1px, transparent 1.4px)';
   722→      root.style.backgroundSize = '16px 16px';
   723→    }
   724→
   725→    var html = pvNavHtml() + pvHeroHtml();
   726→    var alt = false;
   727→    state.order.forEach(function (id) {
   728→      var s = state.sections[id];
   729→      if (!s.on || id === 'hero') return;
   730→      switch (id) {
   731→        case 'countdown': html += pvCountdownHtml(); break;
   732→        case 'welcome':   html += pvSimpleHtml('welcome', s.title, s.text, (alt = !alt)); break;
   733→        case 'story':     html += pvSimpleHtml('story', s.title, s.text, (alt = !alt)); break;
   734→        case 'details':   html += pvDetailsHtml(); break;
   735→        case 'schedule':  html += pvScheduleHtml(); break;
   736→        case 'gallery':   html += pvGalleryHtml(); break;
   737→        case 'rsvp':      html += pvRsvpHtml(); break;
   738→        case 'travel':    html += pvTravelHtml('travel', s.title, s.text, (alt = !alt)); break;
   739→        case 'registry':  html += pvTravelHtml('registry', s.title, s.text, (alt = !alt)); break;
   740→        case 'faq':       html += pvFaqHtml(); break;
   741→        case 'contact':   html += pvContactHtml(); break;
   742→        case 'footer':    html += pvFooterHtml(); break;
   743→      }
   744→    });
   745→    if (!/pv-footer/.test(html)) html += pvFooterHtml();
   746→
   747→    root.innerHTML = html;
   748→    canvas.innerHTML = '';
   749→    canvas.appendChild(root);
   750→
   751→    /* countdown ticker */
   752→    var box = root.querySelector('[data-countdown]');
   753→    if (box) {
   754→      var set = function () {
   755→        var diff = dateObj().getTime() - Date.now();
   756→        var past = diff <= 0;
   757→        if (past) diff = 0;
   758→        var d = Math.floor(diff / 864e5);
   759→        var h = Math.floor(diff % 864e5 / 36e5);
   760→        var m = Math.floor(diff % 36e5 / 6e4);
   761→        var sec = Math.floor(diff % 6e4 / 1e3);
   762→        var q = function (sel, v) {
   763→          var el2 = box.querySelector('[data-cd="' + sel + '"]');
   764→          if (el2) el2.textContent = String(v);
   765→        };
   766→        q('d', d); q('h', h); q('m', m); q('s', sec);
   767→        var lbl = root.querySelector('.pv-count-label');
   768→        if (lbl && past && state.sections.countdown.on) lbl.textContent = 'Just married!';
   769→      };
   770→      set();
   771→      pvTimer = setInterval(set, 1000);
   772→    }
   773→
   774→    if (wrap) wrap.scrollTop = scrollPos;
   775→  }
