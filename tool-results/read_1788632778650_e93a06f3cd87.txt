   372→   372→  App.parseCsv = function (text) {
   373→   373→    var rows = [], row = [], field = '', inQ = false;
   374→   374→    for (var i = 0; i < text.length; i++) {
   375→   375→      var c = text[i];
   376→   376→      if (inQ) {
   377→   377→        if (c === '"') { if (text[i + 1] === '"') { field += '"'; i++; } else inQ = false; }
   378→   378→        else field += c;
   379→   379→      } else if (c === '"') inQ = true;
   380→   380→      else if (c === ',') { row.push(field); field = ''; }
   381→   381→      else if (c === '\n' || c === '\r') {
   382→   382→        if (field !== '' || row.length) { row.push(field); rows.push(row); }
   383→   383→        row = []; field = '';
   384→   384→        if (c === '\r' && text[i + 1] === '\n') i++;
   385→   385→      } else field += c;
   386→   386→    }
   387→   387→    if (field !== '' || row.length) { row.push(field); rows.push(row); }
   388→   388→    return rows;
   389→   389→  };
