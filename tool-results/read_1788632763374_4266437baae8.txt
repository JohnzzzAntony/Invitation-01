   372→   372→  App.parseCsv = function (text) {
   373→   373→    var rows = [], row = [], field = '', inQ = false;
   374→   374→    for (var i = 0; i < text.length; i++) {
   375→   375→      var c = text[i];
   376→   376→      if (inQ) {
   377→   377→        if (c === '"') { if (text[i + 1] === '"') { field += '"'; i++; } else inQ = false; }
   378→   378→        else field += c;
   379→   379→      } else if (c === '"') inQ = true;
   380→   380→      else if (c === ',') { row.push(field); field = ''; }
   381→   381→      else if (c === '\n' || c === '\r') {
   382→   382→        if (field !== '' || row.length) { row.push(field); rows.push(row); }
   383→   383→        row = []; field = '';
   384→   384→        if (c === '\r' && text[i + 1] === '\n') i++;
   385→   385→      } else field += c;
   386→   386→    }
   387→   387→    if (field !== '' || row.length) { row.push(field); rows.push(row); }
   388→   388→    return rows;
   389→   389→  };

   567→   567→  App.bindCountdown = function (el, targetIso, compact) {
   568→   568→    if (!el) return;
   569→   569→    var target = new Date(targetIso).getTime();
   570→   570→    function tick() {
   571→   571→      var diff = Math.max(0, target - Date.now());
   572→   572→      var d = Math.floor(diff / 86400000);
   573→   573→      var h = Math.floor((diff % 86400000) / 3600000);
   574→   574→      var m = Math.floor((diff % 3600000) / 60000);
   575→   575→      var s = Math.floor((diff % 60000) / 1000);
   576→   576→      var cells = [[d, 'Days'], [h, 'Hours'], [m, 'Minutes'], [s, 'Seconds']];
   577→   577→      el.innerHTML = cells.map(function (c) {
   578→   578→        return '<div class="mc-box"><div class="mc-num">' + String(c[0]).padStart(2, '0') + '</div><div class="mc-lab">' + c[1] + '</div></div>';
   579→   579→      }).join('');
   580→   580→    }
   581→   581→    tick();
   582→   582→    var iv = setInterval(tick, 1000);
   583→   583→    App.addCleanup(function () { clearInterval(iv); });
   584→   584→  };
